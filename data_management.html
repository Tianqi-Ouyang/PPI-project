NoteAbout this page

Executable specification of the pipeline. Chunks are not run at render time — they read RPDR extracts that stay local. To run:

knitr::purl("data_management.qmd", output = "cache/data_management.R", documentation = 0)
source("cache/data_management.R")

Helpers live in R/ppi_functions.R. Outputs land in cache/ (gitignored); the aggregate cache/qc.rds is what analysis.qmd renders from.

The analysis has four pieces:

  1. Cohort — the 8,911 patients with no PPI at or before ICI start and at least one creatinine afterwards.
  2. Outcome — AKI, computed from each patient’s serial creatinine after the ICI start date.
  3. Exposure — PPI, time-varying. Every patient is unexposed at ICI start (day 0); a patient who fills a PPI on day k becomes exposed from day k.
  4. Follow-up — to the earliest of AKI, last creatinine, death, or 730 days.

## PPI dispensings. `<=` (baseline) and `>` (initiation) are complementary, so a
## new-user can never also be counted as an initiator.
med <- cached("med_ppi", {
  m <- data.table::fread(rpdr("Med"), sep = "|", quote = "", fill = TRUE,
                         showProgress = FALSE,
                         select = c("EMPI", "Medication_Date", "Medication"))
  m[, Medication_Date := to_date(Medication_Date, "%m/%d/%Y")]
  m[, EMPI := as.character(EMPI)]
  m[grepl(name_regex(PPI_NAMES), Medication, ignore.case = TRUE)][]
})
master_med <- med %>% inner_join(baseline %>% select(EMPI, ICI_Dose_Date), by = "EMPI")

ppi_pre  <- med_baseline(master_med, PPI_NAMES)                       # PPI on/before ICI
ppi_post <- med_first_after(master_med, PPI_NAMES) %>%               # first PPI after ICI
  rename(ppi_first_post_date = first_date)

## Systemic steroids are the key time-varying confounder: in ICI patients they
## are started to treat immune-related adverse events (including interstitial
## nephritis), and a PPI is routinely co-prescribed alongside them. Without this
## term the PPI coefficient absorbs the steroid indication.
med_ster <- cached("med_steroid", {
  m <- data.table::fread(rpdr("Med"), sep = "|", quote = "", fill = TRUE,
                         showProgress = FALSE,
                         select = c("EMPI", "Medication_Date", "Medication"))
  m[, Medication_Date := to_date(Medication_Date, "%m/%d/%Y")]
  m[, EMPI := as.character(EMPI)]
  m[grepl(name_regex(STEROID_NAMES), Medication, ignore.case = TRUE) &
    !grepl(name_regex(STEROID_EXCLUDE), Medication, ignore.case = TRUE)][]
})

ster_post <- med_ster %>%
  inner_join(baseline %>% select(EMPI, ICI_Dose_Date), by = "EMPI") %>%
  med_first_after_excl(STEROID_NAMES, STEROID_EXCLUDE) %>%
  rename(ster_first_post_date = first_date)
Show code 
## Serial creatinine (post-ICI creatinine feeds the AKI outcome; the last
## creatinine at any time defines end of surveillance).
cre <- cached("cre", {
  lab <- data.table::fread(rpdr("Lab"), sep = "|", quote = "", fill = TRUE,
                           showProgress = FALSE,
                           select = c("EMPI", "Seq_Date_Time", "Group_Id", "Result"))
  lab <- lab[Group_Id == "CRE"]
  lab[, `:=`(EMPI = as.character(EMPI),
             cre_date = to_date(Seq_Date_Time, "%m/%d/%Y"),
             cre = parse_lab_numeric(Result))]
  lab[!is.na(cre_date) & !is.na(cre), .(EMPI, cre_date, cre)][]
})

last_cre <- cre %>% group_by(EMPI) %>%
  summarise(last_cre_date = max(cre_date), .groups = "drop")

ledger <- list(); step <- function(lbl, df) {
  ledger[[length(ledger) + 1]] <<- tibble(step = lbl, n = nrow(df)); df }

coh <- baseline %>% step("ICI-CKD analytic master (as saved)", .) %>%
  filter(is.na(Date_Of_Death) | Date_Of_Death >= ICI_Dose_Date) %>%
    step("- death recorded before ICI start", .) %>%
  anti_join(ppi_pre, by = "EMPI") %>%
    step("- PPI on or before ICI (keep new-users)", .) %>%
  left_join(last_cre, by = "EMPI") %>%
  mutate(last_cre_days = as.numeric(last_cre_date - ICI_Dose_Date)) %>%
  filter(!is.na(last_cre_days), last_cre_days > 0) %>%
    step("- no creatinine after ICI (keep ascertainable)", .)

QC$ledger <- bind_rows(ledger) %>% mutate(removed = lag(n) - n)

3b CKD composite outcome (Kavita definition)

The same CKD composite as the Kavita ICI-CKD project: ESRD/kidney transplant after ICI, or a sustained (>=90 day) eGFR < 10, or incident CKD (baseline eGFR >= 60 with a sustained >30% drop to eGFR < 60), or CKD progression (baseline eGFR < 60 sustained). The component flags and their first-event dates are precomputed in the analytic master; make_ckd_composite() combines them exactly as in Final cohort/Kavita cohort 2026 0130.Rmd.

4 Follow-up & time-varying exposure

Show code 
pt <- coh %>%
  left_join(ppi_post,  by = "EMPI") %>%
  left_join(ster_post, by = "EMPI") %>%
  mutate(
    ppi_day    = as.numeric(ppi_first_post_date  - ICI_Dose_Date),
    ster_day   = as.numeric(ster_first_post_date - ICI_Dose_Date),
    death_days = as.numeric(Date_Of_Death - ICI_Dose_Date),
    ## Observation window for the co-timing diagnostics. Unlike fu_days this
    ## does NOT stop at the AKI, so an AKI occurring before the PPI still has
    ## its person-time counted (see mirror_window()).
    obs_days   = pmin(last_cre_days, death_days, CFG$HORIZON, na.rm = TRUE),
    ## Follow-up ends at the earliest of AKI, last creatinine, death, horizon.
    fu_days    = pmin(aki_days, last_cre_days, death_days, CFG$HORIZON, na.rm = TRUE),
    ## Event = AKI within follow-up.
    ev         = as.integer(!is.na(aki_days) & aki_days <= fu_days),
    ## Initiator = starts a PPI while still at risk.
    ppi_initiator = as.integer(!is.na(ppi_day) & ppi_day > 0 & ppi_day < fu_days),

    ## CKD composite -- its own follow-up clock, same time-varying PPI exposure.
    ## Censor at last creatinine / death / horizon (the Kavita censor convention).
    fu_ckd     = pmin(ckd_days, last_cre_days, death_days, CFG$HORIZON, na.rm = TRUE),
    ev_ckd     = as.integer(!is.na(ckd_days) & ckd_days <= fu_ckd),
    ckd_initiator = as.integer(!is.na(ppi_day) & ppi_day > 0 & ppi_day < fu_ckd)
  ) %>%
  filter(fu_days > 0)

## Impute the handful of missing baseline eGFR at the patient level (an
## adjusted model would otherwise silently drop those rows).
if (any(is.na(pt$eGFR_CRE_baseline)))
  pt$eGFR_CRE_baseline[is.na(pt$eGFR_CRE_baseline)] <-
    median(pt$eGFR_CRE_baseline, na.rm = TRUE)

QC$cohort <- tibble(
  n            = nrow(pt),
  ppi_initiators = sum(pt$ppi_initiator),
  never_ppi    = sum(pt$ppi_initiator == 0),
  aki_events   = sum(pt$ev),
  person_years = round(sum(pt$fu_days) / 365.25, 1),
  median_fu_days = median(pt$fu_days)
)

## Steroid co-initiation -- the imbalance the time-varying term exists to absorb.
QC$steroid <- pt %>%
  group_by(group = ifelse(ppi_initiator == 1, "Started PPI", "Never PPI")) %>%
  summarise(n = dplyr::n(),
            started_steroid = sum(!is.na(ster_day) & ster_day <= obs_days),
            pct = round(100 * mean(!is.na(ster_day) & ster_day <= obs_days), 1),
            .groups = "drop")

## One row per exposure interval: unexposed (0 -> ppi_day) then exposed
## (ppi_day -> end). Surv(tstart, tstop, event) ~ ppi.
cp <- build_cp_ppi(
  pt %>% transmute(id = EMPI, fu_days, ev, ppi_day,
                   age_ici, male, Race, dm, htn, eGFR_CRE_baseline)
)

## Person-time and crude rate by exposure state.
QC$person_time <- cp %>%
  mutate(days = tstop - tstart) %>%
  group_by(exposure = ifelse(ppi == 1, "On PPI (after initiation)", "Not on PPI")) %>%
  summarise(person_years = round(sum(days) / 365.25, 0),
            aki_events   = sum(event), .groups = "drop") %>%
  mutate(rate_per_100py = round(100 * aki_events / person_years, 1))

## Sanity: exactly the cohort's patients and events survive the split.
stopifnot(dplyr::n_distinct(cp$id) == nrow(pt), sum(cp$event) == sum(pt$ev))

QC$cohort_ckd <- tibble(

QC$person_time_ckd <- cp_ckd %>%
  mutate(days = tstop - tstart) %>%
  group_by(exposure = ifelse(ppi == 1, "On PPI (after initiation)", "Not on PPI")) %>%
  summarise(person_years = round(sum(days) / 365.25, 0),

  mutate(rate_per_100py = round(100 * ckd_events / person_years, 1))
